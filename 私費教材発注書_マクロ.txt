' ===== ここから下を全部コピーして貼り付ける =====
Option Explicit

Private Const FIRST_ROW As Long = 21    ' 発注書の教材1行目
Private Const ROWS_PER_PAGE As Long = 7 ' 発注書1枚に入る教材の数
Private Const T_FIRST As Long = 19      ' 発注シートの一覧の1行目
Private Const T_LAST As Long = 48       ' 発注シートの一覧の最後の行
Private Const KEY_CELL As String = "O17" ' 「教材を出す」を押したときの入力を覚えておく場所
Private Const CK_FIRST As Long = 4      ' 発注済確認リストの1行目

' ===== ボタン：教材を出す =====
Sub 教材を出す()
    Dim wsIn As Worksheet, wsList As Worksheet
    Set wsIn = ThisWorkbook.Worksheets("発注")
    Set wsList = ThisWorkbook.Worksheets("教材リスト")

    Dim company As String, kai As Long, isTennyu As Boolean, tNum(1 To 5) As Double
    If Not ReadInput(company, kai, isTennyu, tNum) Then Exit Sub

    Dim clsNum(1 To 5) As Double, j As Long
    For j = 1 To 5
        clsNum(j) = Val(ThisWorkbook.Worksheets("設定").Cells(7 + j, 3).Value)
    Next j

    ' 一覧を空にする（D列の式は残す）
    wsIn.Range("B" & T_FIRST & ":C" & T_LAST).ClearContents
    wsIn.Range("E" & T_FIRST & ":O" & T_LAST).ClearContents

    Dim lastRow As Long, r As Long, t As Long, per As Double, howTo As String, baseName As String
    lastRow = wsList.Cells(wsList.Rows.Count, 3).End(xlUp).Row
    t = T_FIRST
    For r = 4 To lastRow
        baseName = Trim(CStr(wsList.Cells(r, 3).Value))
        howTo = Trim(CStr(wsList.Cells(r, 5).Value))
        If Trim(CStr(wsList.Cells(r, 2).Value)) = company And baseName <> "" Then
            If IsThisTime(howTo, kai, isTennyu) Then
                If t > T_LAST Then
                    MsgBox "教材が多すぎて、一覧に全部入りません（" & (T_LAST - T_FIRST + 1) & "個まで）。", vbExclamation
                    Exit For
                End If
                per = Val(wsList.Cells(r, 6).Value)
                If per <= 0 Then per = 1
                wsIn.Cells(t, 2).Value = "○"
                wsIn.Cells(t, 3).Value = baseName
                wsIn.Cells(t, 5).Value = howTo
                wsIn.Cells(t, 6).Value = wsList.Cells(r, 4).Value
                wsIn.Cells(t, 7).Value = per
                For j = 1 To 5
                    If isTennyu Then
                        If tNum(j) > 0 Then wsIn.Cells(t, 7 + j).Value = tNum(j) * per
                    ElseIf clsNum(j) > 0 Then
                        wsIn.Cells(t, 7 + j).Value = clsNum(j) * per
                    End If
                Next j
                wsIn.Cells(t, 13).Value = wsList.Cells(r, 7).Value
                wsIn.Cells(t, 14).Value = wsList.Cells(r, 8).Value
                wsIn.Cells(t, 15).Value = r
                t = t + 1
            End If
        End If
    Next r

    wsIn.Range(KEY_CELL).Value = InputKey()
    UpdateCheckList

    If t = T_FIRST Then
        MsgBox "「" & company & "」の" & kai & "回目に頼む教材は、教材リストにありません。" & vbCrLf & _
               "急に頼むものがあれば、一覧のあいている行に書いてください。", vbInformation
    Else
        MsgBox (t - T_FIRST) & "個の教材を出しました。" & vbCrLf & _
               "頼まないものは「頼む」の○を消してから、「発注書をつくる」を押してください。", vbInformation
    End If
End Sub

' ===== ボタン：発注書をつくる =====
Sub 発注書をつくる()
    Dim wsIn As Worksheet, wsSet As Worksheet, wsList As Worksheet, wsTmp As Worksheet
    Set wsIn = ThisWorkbook.Worksheets("発注")
    Set wsSet = ThisWorkbook.Worksheets("設定")
    Set wsList = ThisWorkbook.Worksheets("教材リスト")
    Set wsTmp = ThisWorkbook.Worksheets("発注書ひな形")

    Dim company As String, kai As Long, isTennyu As Boolean, tNum(1 To 5) As Double
    If Not ReadInput(company, kai, isTennyu, tNum) Then Exit Sub

    If CStr(wsIn.Range(KEY_CELL).Value) <> InputKey() Then
        MsgBox "会社名・何回目・転入生の分かどうかが変わっています。先に「教材を出す」を押してください。", vbExclamation
        Exit Sub
    End If

    Dim orderDate As Date, deliver As String
    If IsDate(wsIn.Range("D7").Value) Then
        orderDate = CDate(wsIn.Range("D7").Value)
    Else
        orderDate = Date
    End If
    deliver = CStr(wsIn.Range("D8").Value)

    Dim clsName(1 To 5) As String, clsNum(1 To 5) As Double, j As Long
    For j = 1 To 5
        clsName(j) = Trim(CStr(wsSet.Cells(7 + j, 2).Value))
        clsNum(j) = Val(wsSet.Cells(7 + j, 3).Value)
    Next j

    ' --- 一覧から「頼む」に何か入っている行を集める ---
    Dim cnt As Long, t As Long, n As Long, per As Double, hasNum As Boolean
    cnt = T_LAST - T_FIRST + 1
    Dim itRow() As Long, itName() As String, itQty() As Variant
    ReDim itRow(1 To cnt), itName(1 To cnt), itQty(1 To cnt, 1 To 5)

    For t = T_FIRST To T_LAST
        If Trim(CStr(wsIn.Cells(t, 2).Value)) <> "" And Trim(CStr(wsIn.Cells(t, 3).Value)) <> "" Then
            n = n + 1
            itRow(n) = t
            itName(n) = NameWithSuffix(Trim(CStr(wsIn.Cells(t, 3).Value)), Trim(CStr(wsIn.Cells(t, 5).Value)), kai)
            per = Val(wsIn.Cells(t, 7).Value)
            If per <= 0 Then per = 1
            hasNum = False
            For j = 1 To 5
                If Trim(CStr(wsIn.Cells(t, 7 + j).Value)) <> "" Then hasNum = True
            Next j
            For j = 1 To 5
                If hasNum Then
                    itQty(n, j) = wsIn.Cells(t, 7 + j).Value
                ElseIf isTennyu Then
                    If tNum(j) > 0 Then itQty(n, j) = tNum(j) * per
                ElseIf clsName(j) <> "" And clsNum(j) > 0 Then
                    itQty(n, j) = clsNum(j) * per
                End If
            Next j
        End If
    Next t

    If n = 0 Then
        MsgBox "「頼む」に○がついている教材がありません。", vbExclamation
        Exit Sub
    End If

    ' --- 発注書をつくる（7つごとに1枚） ---
    Dim pages As Long, p As Long, k As Long, rw As Long, idx As Long
    Dim ws As Worksheet, firstWs As Worksheet, baseName As String, colLetters As Variant
    colLetters = Array("AA", "AD", "AG", "AJ", "AM")
    pages = (n + ROWS_PER_PAGE - 1) \ ROWS_PER_PAGE
    baseName = ReiwaText(orderDate, "、") & "月 " & CleanName(company)

    Application.ScreenUpdating = False
    For p = 1 To pages
        wsTmp.Copy After:=ThisWorkbook.Sheets(ThisWorkbook.Sheets.Count)
        Set ws = ThisWorkbook.Sheets(ThisWorkbook.Sheets.Count)
        If pages = 1 Then
            ws.Name = UniqueName(baseName, "")
        Else
            ws.Name = UniqueName(baseName, " " & p & "枚目")
        End If
        If p = 1 Then Set firstWs = ws

        ws.Range("G6").Value = company
        ws.Range("AZ6").Value = Month(orderDate)
        ws.Range("BD6").Value = Day(orderDate)
        ws.Range("AU10").Value = wsSet.Range("C5").Value
        ws.Range("D12").Value = wsSet.Range("C4").Value
        ws.Range("O14").Value = pages
        ws.Range("AF14").Value = deliver
        For j = 1 To 5
            ws.Range(colLetters(j - 1) & "20").Value = clsName(j)
        Next j

        For k = 1 To ROWS_PER_PAGE
            rw = FIRST_ROW + (k - 1) * 2
            idx = (p - 1) * ROWS_PER_PAGE + k
            If idx <= n Then
                ws.Range("C" & rw).Value = itName(idx)
                ws.Range("V" & rw).Value = wsIn.Cells(itRow(idx), 6).Value
                For j = 1 To 5
                    ws.Range(colLetters(j - 1) & rw).Value = itQty(idx, j)
                Next j
                ws.Range("AV" & rw).Value = wsIn.Cells(itRow(idx), 13).Value
                ws.Range("BB" & rw).Value = wsIn.Cells(itRow(idx), 14).Value
            End If
        Next k

        ' 冊数合計の式を入れ直し、教材の欄の取り消し線を外す
        For k = 1 To ROWS_PER_PAGE
            rw = FIRST_ROW + (k - 1) * 2
            ws.Range("AP" & rw).Formula = "=IF(SUM(AA" & rw & ":AO" & (rw + 1) & ")>0,SUM(AA" & rw & ":AO" & (rw + 1) & "),"""")"
        Next k
        ws.Range("C" & FIRST_ROW & ":BG" & (FIRST_ROW + ROWS_PER_PAGE * 2 - 1)).Font.Strikethrough = False
    Next p
    Application.ScreenUpdating = True

    firstWs.Activate
    MsgBox pages & "枚の発注書をつくりました。" & vbCrLf & _
           "教師用冊数は、発注書に直接入れてください。", vbInformation

    ' --- 急に頼んだ教材を、教材リストに入れるか聞く ---
    Dim newRow As Long, howTo As String
    For idx = 1 To n
        t = itRow(idx)
        If Trim(CStr(wsIn.Cells(t, 15).Value)) = "" Then
            If MsgBox("「" & wsIn.Cells(t, 3).Value & "」を教材リストに入れますか？" & vbCrLf & _
                      "（入れると、次からは「教材を出す」で出てきます）", vbYesNo + vbQuestion) = vbYes Then
                newRow = wsList.Cells(wsList.Rows.Count, 3).End(xlUp).Row + 1
                If newRow < 4 Then newRow = 4
                howTo = Trim(CStr(wsIn.Cells(t, 5).Value))
                If howTo = "" Then howTo = "年1回"
                wsList.Cells(newRow, 2).Value = company
                wsList.Cells(newRow, 3).Value = Trim(CStr(wsIn.Cells(t, 3).Value))
                wsList.Cells(newRow, 4).Value = wsIn.Cells(t, 6).Value
                wsList.Cells(newRow, 5).Value = howTo
                per = Val(wsIn.Cells(t, 7).Value)
                If per <= 0 Then per = 1
                wsList.Cells(newRow, 6).Value = per
                wsList.Cells(newRow, 7).Value = wsIn.Cells(t, 13).Value
                wsList.Cells(newRow, 8).Value = wsIn.Cells(t, 14).Value
                wsIn.Cells(t, 15).Value = newRow
            End If
        End If
    Next idx

    ' --- 発注済確認リストに書く ---
    UpdateCheckList
    For idx = 1 To n
        MarkOrdered company, itName(idx), orderDate, isTennyu
    Next idx
End Sub

' ===== ボタン：確認リストを新しくする（発注済確認リストのシートで使う） =====
Sub 確認リストを新しくする()
    UpdateCheckList
    MsgBox "発注済確認リストを、教材リストに合わせて新しくしました。", vbInformation
End Sub

' 教材リストから発注済確認リストをつくり直す。○・発注日・備考は消さずに残す
Private Sub UpdateCheckList()
    Dim wsList As Worksheet, wsCk As Worksheet
    Set wsList = ThisWorkbook.Worksheets("教材リスト")
    Set wsCk = ThisWorkbook.Worksheets("発注済確認リスト")

    ' 今の確認リストを覚えておく
    Dim oldLast As Long, oldN As Long, r As Long, i As Long
    oldLast = wsCk.Cells(wsCk.Rows.Count, 6).End(xlUp).Row
    If oldLast < CK_FIRST Then oldLast = CK_FIRST - 1
    oldN = oldLast - CK_FIRST + 1
    Dim oKey() As String, oName() As String, oMark() As Variant, oDate() As Variant, oNote() As Variant, oUsed() As Boolean
    If oldN > 0 Then
        ReDim oKey(1 To oldN), oName(1 To oldN), oMark(1 To oldN), oDate(1 To oldN), oNote(1 To oldN), oUsed(1 To oldN)
        For i = 1 To oldN
            r = CK_FIRST + i - 1
            oKey(i) = CStr(wsCk.Cells(r, 6).Value)
            oName(i) = CStr(wsCk.Cells(r, 3).Value)
            oMark(i) = wsCk.Cells(r, 2).Value
            oDate(i) = wsCk.Cells(r, 4).Value
            oNote(i) = wsCk.Cells(r, 5).Value
        Next i
    End If

    ' 教材リストから、上刊・下刊や1学期〜3学期に分けた行をつくる
    Dim lastRow As Long, company As String, baseName As String, howTo As String
    Dim names(1 To 3) As String, m As Long, k As Long
    Dim nKey() As String, nName() As String, nN As Long
    lastRow = wsList.Cells(wsList.Rows.Count, 3).End(xlUp).Row
    ReDim nKey(1 To (lastRow + 1) * 3 + oldN), nName(1 To (lastRow + 1) * 3 + oldN)
    For r = 4 To lastRow
        company = Trim(CStr(wsList.Cells(r, 2).Value))
        baseName = Trim(CStr(wsList.Cells(r, 3).Value))
        howTo = Trim(CStr(wsList.Cells(r, 5).Value))
        If baseName <> "" Then
            Select Case HowToKind(howTo)
                Case 2
                    m = 2: names(1) = baseName & " 上刊": names(2) = baseName & " 下刊"
                Case 3
                    m = 3: names(1) = baseName & " 1学期": names(2) = baseName & " 2学期": names(3) = baseName & " 3学期"
                Case Else
                    m = 1: names(1) = baseName
            End Select
            For k = 1 To m
                If FindIndex(nKey, nN, company & "|" & names(k)) = 0 Then
                    nN = nN + 1
                    nKey(nN) = company & "|" & names(k)
                    nName(nN) = names(k)
                End If
            Next k
        End If
    Next r

    ' 教材リストにないが、○や備考が入っている行は残す
    For i = 1 To oldN
        If oKey(i) <> "" And FindIndex(nKey, nN, oKey(i)) = 0 Then
            If Trim(CStr(oMark(i))) <> "" Or Trim(CStr(oNote(i))) <> "" Then
                nN = nN + 1
                nKey(nN) = oKey(i)
                nName(nN) = oName(i)
            End If
        End If
    Next i

    ' 書き直す
    If oldN > 0 Then wsCk.Range("B" & CK_FIRST & ":F" & oldLast).ClearContents
    For k = 1 To nN
        r = CK_FIRST + k - 1
        wsCk.Cells(r, 3).Value = nName(k)
        wsCk.Cells(r, 6).Value = nKey(k)
        If oldN > 0 Then
            i = FindIndex(oKey, oldN, nKey(k))
            If i > 0 Then
                wsCk.Cells(r, 2).Value = oMark(i)
                wsCk.Cells(r, 4).Value = oDate(i)
                wsCk.Cells(r, 5).Value = oNote(i)
            End If
        End If
    Next k
End Sub

' 発注した教材に○と発注日をつける。転入生の分は備考に書き足す
Private Sub MarkOrdered(ByVal company As String, ByVal itemName As String, ByVal orderDate As Date, ByVal isTennyu As Boolean)
    Dim wsCk As Worksheet, lastRow As Long, r As Long, found As Long, key As String
    Set wsCk = ThisWorkbook.Worksheets("発注済確認リスト")
    key = company & "|" & itemName
    lastRow = wsCk.Cells(wsCk.Rows.Count, 6).End(xlUp).Row
    For r = CK_FIRST To lastRow
        If CStr(wsCk.Cells(r, 6).Value) = key Then found = r: Exit For
    Next r
    If found = 0 Then
        ' 教材リストに入れなかった教材は、いちばん下に足す
        found = lastRow + 1
        If found < CK_FIRST Then found = CK_FIRST
        wsCk.Cells(found, 3).Value = itemName
        wsCk.Cells(found, 6).Value = key
    End If
    If isTennyu Then
        If Trim(CStr(wsCk.Cells(found, 5).Value)) = "" Then
            wsCk.Cells(found, 5).Value = "転入 " & DateText(orderDate)
        Else
            wsCk.Cells(found, 5).Value = wsCk.Cells(found, 5).Value & "、転入 " & DateText(orderDate)
        End If
    Else
        wsCk.Cells(found, 2).Value = "○"
        wsCk.Cells(found, 4).Value = DateText(orderDate)
    End If
End Sub

' 配列の1〜nの中から同じ文字を探す。見つからなければ 0
Private Function FindIndex(arr() As String, ByVal n As Long, ByVal s As String) As Long
    Dim i As Long
    For i = 1 To n
        If arr(i) = s Then FindIndex = i: Exit Function
    Next i
End Function

' 「R8」＋区切り＋月 をつくる（例：R8、4／R8.4）
Private Function ReiwaText(ByVal d As Date, ByVal sep As String) As String
    ReiwaText = "R" & (Year(d) - 2018) & sep & Month(d)
End Function

' 「R8.4.10」の形の日付
Private Function DateText(ByVal d As Date) As String
    DateText = ReiwaText(d, ".") & "." & Day(d)
End Function

' 発注シートの上の欄を読む。足りないときはメッセージを出して False を返す
Private Function ReadInput(company As String, kai As Long, isTennyu As Boolean, tNum() As Double) As Boolean
    Dim wsIn As Worksheet, j As Long, total As Double
    Set wsIn = ThisWorkbook.Worksheets("発注")
    ReadInput = False
    company = Trim(CStr(wsIn.Range("D4").Value))
    kai = Val(Left(CStr(wsIn.Range("D5").Value), 1))
    isTennyu = (Trim(CStr(wsIn.Range("D6").Value)) = "はい")
    If company = "" Then
        MsgBox "「会社名」を選んでください。", vbExclamation
        Exit Function
    End If
    If kai < 1 Or kai > 3 Then
        MsgBox "「何回目」を選んでください。", vbExclamation
        Exit Function
    End If
    For j = 1 To 5
        tNum(j) = 0
        If isTennyu Then tNum(j) = Val(wsIn.Cells(10 + j, 4).Value)
        total = total + tNum(j)
    Next j
    If isTennyu And total <= 0 Then
        MsgBox "転入生の分のときは、転入生が来た組に人数を入れてください。", vbExclamation
        Exit Function
    End If
    ReadInput = True
End Function

' 会社名・何回目・転入生の分か・転入の人数をまとめた文字（入力が変わったか調べるため）
Private Function InputKey() As String
    Dim wsIn As Worksheet, j As Long, s As String
    Set wsIn = ThisWorkbook.Worksheets("発注")
    s = Trim(CStr(wsIn.Range("D4").Value)) & "|" & CStr(wsIn.Range("D5").Value) & "|" & CStr(wsIn.Range("D6").Value)
    If Trim(CStr(wsIn.Range("D6").Value)) = "はい" Then
        For j = 1 To 5
            s = s & "|" & CStr(wsIn.Cells(10 + j, 4).Value)
        Next j
    End If
    InputKey = s
End Function

' 頼み方の種類：1＝年1回、2＝上刊・下刊、3＝1学期・2学期・3学期
Private Function HowToKind(ByVal howTo As String) As Long
    Select Case howTo
        Case "上刊・下刊"
            HowToKind = 2
        Case "1学期・2学期・3学期", "１学期・２学期・３学期", "一学期・二学期・三学期"
            HowToKind = 3
        Case Else
            HowToKind = 1
    End Select
End Function

' 今回の発注で出すものか
Private Function IsThisTime(ByVal howTo As String, ByVal kai As Long, ByVal isTennyu As Boolean) As Boolean
    Select Case HowToKind(howTo)
        Case 2
            IsThisTime = (kai <= 2) Or isTennyu
        Case 3
            IsThisTime = True
        Case Else
            IsThisTime = (kai = 1) Or isTennyu
    End Select
End Function

' 教材名の後ろに「上刊」「1学期」などをつける
Private Function NameWithSuffix(ByVal baseName As String, ByVal howTo As String, ByVal kai As Long) As String
    Select Case HowToKind(howTo)
        Case 2
            If kai = 1 Then
                NameWithSuffix = baseName & " 上刊"
            Else
                NameWithSuffix = baseName & " 下刊"
            End If
        Case 3
            NameWithSuffix = baseName & " " & kai & "学期"
        Case Else
            NameWithSuffix = baseName
    End Select
End Function

' シート名に使えない文字を取る
Private Function CleanName(ByVal s As String) As String
    Dim c As Variant
    For Each c In Array(":", "\", "/", "?", "*", "[", "]")
        s = Replace(s, c, "")
    Next c
    CleanName = s
End Function

' 同じ名前のシートがあれば (2)(3)… をつける
Private Function UniqueName(ByVal baseName As String, ByVal suffix As String) As String
    Dim nm As String, i As Long
    i = 1
    Do
        If i = 1 Then
            nm = baseName & suffix
        Else
            nm = baseName & suffix & "(" & i & ")"
        End If
        If Len(nm) > 31 Then nm = Left(baseName, 31 - Len(nm) + Len(baseName)) & Mid(nm, Len(baseName) + 1)
        If Not SheetExists(nm) Then Exit Do
        i = i + 1
    Loop
    UniqueName = nm
End Function

Private Function SheetExists(ByVal nm As String) As Boolean
    Dim ws As Object
    For Each ws In ThisWorkbook.Sheets
        If ws.Name = nm Then SheetExists = True: Exit Function
    Next ws
End Function
