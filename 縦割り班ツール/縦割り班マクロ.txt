Option Explicit

'====================================================
' 縦割り班づくり
' 「入力」シートの人数から、1松〜6松の班表を作ります。
' 教師用（暗語の欄あり）と児童用（暗語の欄なし）を作ります。
'====================================================

Const PINK As Long = 15123199   ' ピンク RGB(255,194,230)

Sub 縦割り班を作る()
    Dim wsIn As Worksheet
    Dim cls As String
    Dim boys(1 To 6) As Long, girls(1 To 6) As Long
    Dim g As Long, n As Long, h As Long, u As Long, k As Long
    Dim ws As Worksheet, hasOld As Boolean

    On Error Resume Next
    Set wsIn = ThisWorkbook.Worksheets("入力")
    On Error GoTo 0
    If wsIn Is Nothing Then
        MsgBox "「入力」シートが見つかりません。", vbExclamation
        Exit Sub
    End If

    cls = Trim(CStr(wsIn.Range("C3").Value))
    If cls = "" Then
        MsgBox "クラス名（松・竹・梅・月）を入れてください。", vbExclamation
        Exit Sub
    End If

    n = 0
    For g = 1 To 6
        boys(g) = Val(wsIn.Cells(5 + g, 3).Value)
        girls(g) = Val(wsIn.Cells(5 + g, 4).Value)
        If g = 1 Or boys(g) + girls(g) < n Then n = boys(g) + girls(g)
    Next g
    If n <= 0 Then
        MsgBox "どの学年にも1人以上の人数を入れてください。", vbExclamation
        Exit Sub
    End If

    '--- 前に作った班表があれば消す ---
    For Each ws In ThisWorkbook.Worksheets
        If Right(ws.Name, 5) = "（教師用）" Or Right(ws.Name, 5) = "（児童用）" Then hasOld = True
    Next ws
    If hasOld Then
        If MsgBox("前に作った班表を消して、作り直します。" & vbCrLf & _
                  "入れた名前も消えます。よろしいですか？", vbYesNo + vbExclamation) = vbNo Then Exit Sub
        Application.DisplayAlerts = False
        For k = ThisWorkbook.Worksheets.Count To 1 Step -1
            Set ws = ThisWorkbook.Worksheets(k)
            If Right(ws.Name, 5) = "（教師用）" Or Right(ws.Name, 5) = "（児童用）" Then ws.Delete
        Next k
        Application.DisplayAlerts = True
    End If

    '--- 1松〜6松の班の数（高学年から増やす） ---
    Dim unitCnt(1 To 6) As Long
    For u = 1 To 6
        unitCnt(u) = n \ 6
        If u > 6 - (n Mod 6) Then unitCnt(u) = unitCnt(u) + 1
    Next u

    Dim hUnit() As Long, hNo() As Long
    ReDim hUnit(1 To n): ReDim hNo(1 To n)
    h = 0
    For u = 1 To 6
        For k = 1 To unitCnt(u)
            h = h + 1
            hUnit(h) = u
            hNo(h) = k
        Next k
    Next u

    '--- 子どもを班に割り振る（男女・学年のバランス） ---
    Dim mB() As Long, mW() As Long, gT() As Long, bT() As Long, tT() As Long
    ReDim mB(1 To n, 1 To 6): ReDim mW(1 To n, 1 To 6)
    ReDim gT(1 To n): ReDim bT(1 To n): ReDim tT(1 To n)

    Dim remW As Long, remB As Long, isGirl As Boolean
    Dim best As Long, k1 As Long, k2 As Long, k3 As Long
    Dim b1 As Long, b2 As Long, b3 As Long
    For g = 6 To 1 Step -1
        remW = girls(g): remB = boys(g)
        Do While remW + remB > 0
            isGirl = (remW > 0 And remW >= remB)
            best = 0
            For h = 1 To n
                k1 = mB(h, g) + mW(h, g)
                If isGirl Then k2 = gT(h) Else k2 = bT(h)
                k3 = tT(h)
                If best = 0 Then
                    best = h: b1 = k1: b2 = k2: b3 = k3
                ElseIf k1 < b1 Or (k1 = b1 And k2 < b2) Or (k1 = b1 And k2 = b2 And k3 < b3) Then
                    best = h: b1 = k1: b2 = k2: b3 = k3
                End If
            Next h
            tT(best) = tT(best) + 1
            If isGirl Then
                mW(best, g) = mW(best, g) + 1: gT(best) = gT(best) + 1: remW = remW - 1
            Else
                mB(best, g) = mB(best, g) + 1: bT(best) = bT(best) + 1: remB = remB - 1
            End If
        Loop
    Next g

    '--- シートを作る ---
    Application.ScreenUpdating = False
    Dim wsT As Worksheet, wsC As Worksheet, lastSh As Worksheet
    Dim tName As String, cName As String
    Dim firstH As Long, lastH As Long, maxSize As Long, lines As Long

    Set lastSh = wsIn
    For u = 1 To 6
        If unitCnt(u) > 0 Then
            tName = u & cls & "（教師用）"
            Set wsT = ThisWorkbook.Worksheets.Add(After:=lastSh)
            wsT.Name = tName
            Set lastSh = wsT
        End If
    Next u
    For u = 1 To 6
        If unitCnt(u) > 0 Then
            cName = u & cls & "（児童用）"
            Set wsC = ThisWorkbook.Worksheets.Add(After:=lastSh)
            wsC.Name = cName
            Set lastSh = wsC
        End If
    Next u

    For u = 1 To 6
        If unitCnt(u) > 0 Then
            tName = u & cls & "（教師用）"
            cName = u & cls & "（児童用）"
            Set wsT = ThisWorkbook.Worksheets(tName)
            Set wsC = ThisWorkbook.Worksheets(cName)

            firstH = 0: maxSize = 0
            For h = 1 To n
                If hUnit(h) = u Then
                    If firstH = 0 Then firstH = h
                    lastH = h
                    If tT(h) > maxSize Then maxSize = tT(h)
                End If
            Next h
            lines = (maxSize + 2) \ 3

            Call 表のわく(wsT, u & cls, "", True)
            Call 表のわく(wsC, u & cls, tName, False)

            Dim r As Long
            r = 2
            For h = firstH To lastH
                Call 班をかく(wsT, wsC, tName, r, hNo(h), lines, mB, mW, h)
                r = r + lines * 2
            Next h

            Call 印刷の設定(wsT, r - 1)
            Call 印刷の設定(wsC, r - 1)
        End If
    Next u

    wsIn.Activate
    Application.ScreenUpdating = True
    MsgBox "できました。班は全部で " & n & " 班です。" & vbCrLf & _
           "教師用シートに名前を入れると、児童用シートにも入ります。", vbInformation
End Sub

'--- 列の幅・見出し ---
Private Sub 表のわく(ws As Worksheet, title As String, tName As String, isTeacher As Boolean)
    Dim s As Long, c As Long
    ws.Cells.Font.Name = "ＭＳ ゴシック"
    ws.Cells.Font.Size = 11
    ws.Columns(1).ColumnWidth = 5
    For s = 0 To 2
        c = 2 + s * 3
        ws.Columns(c).ColumnWidth = 4
        ws.Columns(c + 1).ColumnWidth = 15
        ws.Columns(c + 2).ColumnWidth = 3.5
    Next s

    ws.Range("A1:D1").Merge
    ws.Range("A1").Value = title
    ws.Range("A1").Font.Size = 16
    ws.Range("A1").Font.Bold = True
    ws.Range("E1:F1").Merge
    ws.Range("E1").Value = "担任名"
    ws.Range("E1").HorizontalAlignment = xlRight
    ws.Range("G1:J1").Merge
    ws.Range("G1").Font.Size = 14
    If isTeacher Then
        ws.Range("G1").Interior.Color = RGB(255, 255, 204)
    Else
        ws.Range("G1").Formula = "=IF('" & tName & "'!G1="""",""""," & "'" & tName & "'!G1)"
    End If
    Call 線(ws.Range("G1:J1"), xlEdgeBottom, xlContinuous)
    ws.Rows(1).RowHeight = 28
End Sub

'--- 1つの班をかく ---
Private Sub 班をかく(wsT As Worksheet, wsC As Worksheet, tName As String, r As Long, _
                    banNo As Long, lines As Long, mB() As Long, mW() As Long, h As Long)
    Dim gr(1 To 60) As Long, gl(1 To 60) As Boolean
    Dim cnt As Long, g As Long, k As Long, j As Long
    Dim rF As Long, rN As Long, c As Long
    Dim nameAdr As String
    Dim wsx As Variant

    '学年の小さい順にならべる（同じ学年は男子→女子）
    cnt = 0
    For g = 1 To 6
        For k = 1 To mB(h, g)
            cnt = cnt + 1: gr(cnt) = g: gl(cnt) = False
        Next k
        For k = 1 To mW(h, g)
            cnt = cnt + 1: gr(cnt) = g: gl(cnt) = True
        Next k
    Next g

    For Each wsx In Array(wsT, wsC)
        With wsx.Range(wsx.Cells(r, 1), wsx.Cells(r + lines * 2 - 1, 1))
            .Merge
            .Value = banNo
            .HorizontalAlignment = xlCenter
            .VerticalAlignment = xlCenter
            .Font.Size = 14
            .Font.Bold = True
        End With
        For j = 0 To lines - 1
            wsx.Rows(r + j * 2).RowHeight = 13
            wsx.Rows(r + j * 2 + 1).RowHeight = 26
        Next j
    Next wsx

    For j = 0 To lines * 3 - 1
        rF = r + (j \ 3) * 2
        rN = rF + 1
        c = 2 + (j Mod 3) * 3
        nameAdr = wsT.Cells(rN, c + 1).Address(False, False)

        For Each wsx In Array(wsT, wsC)
            With wsx.Range(wsx.Cells(rF, c), wsx.Cells(rN, c))
                .Merge
                .HorizontalAlignment = xlCenter
                .VerticalAlignment = xlCenter
                If j < cnt Then .Value = gr(j + 1)
            End With
        Next wsx

        '教師用：ふりがな／名前／暗語
        With wsT.Cells(rF, c + 1)
            .Formula = "=PHONETIC(" & nameAdr & ")"
            .Font.Size = 8
            .HorizontalAlignment = xlCenter
            .VerticalAlignment = xlBottom
        End With
        With wsT.Cells(rN, c + 1)
            .Font.Size = 14
            .HorizontalAlignment = xlCenter
            .VerticalAlignment = xlCenter
            .ShrinkToFit = True
            On Error Resume Next
            .Phonetic.CharacterType = xlHiragana
            On Error GoTo 0
        End With
        With wsT.Range(wsT.Cells(rF, c + 2), wsT.Cells(rN, c + 2))
            .Merge
            .HorizontalAlignment = xlCenter
            .VerticalAlignment = xlCenter
            .Font.Size = 12
        End With

        '児童用：ふりがな／名前（教師用から自動で入る）
        With wsC.Range(wsC.Cells(rF, c + 1), wsC.Cells(rF, c + 2))
            .Merge
            .Formula = "=PHONETIC('" & tName & "'!" & nameAdr & ")"
            .Font.Size = 8
            .HorizontalAlignment = xlCenter
            .VerticalAlignment = xlBottom
        End With
        With wsC.Range(wsC.Cells(rN, c + 1), wsC.Cells(rN, c + 2))
            .Merge
            .Formula = "=IF('" & tName & "'!" & nameAdr & "="""",""""," & "'" & tName & "'!" & nameAdr & ")"
            .Font.Size = 14
            .HorizontalAlignment = xlCenter
            .VerticalAlignment = xlCenter
            .ShrinkToFit = True
        End With

        '女子はピンク
        If j < cnt Then
            If gl(j + 1) Then
                wsT.Range(wsT.Cells(rF, c + 1), wsT.Cells(rN, c + 1)).Interior.Color = PINK
                wsC.Range(wsC.Cells(rF, c + 1), wsC.Cells(rN, c + 2)).Interior.Color = PINK
            End If
        End If

        '線をひく
        For Each wsx In Array(wsT, wsC)
            wsx.Range(wsx.Cells(rF, c), wsx.Cells(rN, c + 2)).BorderAround xlContinuous, xlThin
            Call 線(wsx.Range(wsx.Cells(rF, c), wsx.Cells(rN, c)), xlEdgeRight, xlContinuous)
        Next wsx
        Call 線(wsT.Cells(rF, c + 1), xlEdgeBottom, xlDot)
        Call 線(wsC.Range(wsC.Cells(rF, c + 1), wsC.Cells(rF, c + 2)), xlEdgeBottom, xlDot)
        Call 線(wsT.Range(wsT.Cells(rF, c + 2), wsT.Cells(rN, c + 2)), xlEdgeLeft, xlContinuous)
    Next j

    '班のまわりを太い線で囲む
    For Each wsx In Array(wsT, wsC)
        wsx.Range(wsx.Cells(r, 1), wsx.Cells(r + lines * 2 - 1, 10)).BorderAround xlContinuous, xlMedium
        Call 線(wsx.Range(wsx.Cells(r, 1), wsx.Cells(r + lines * 2 - 1, 1)), xlEdgeRight, xlContinuous)
    Next wsx
End Sub

'--- 細い線をひく ---
Private Sub 線(rng As Range, edge As Long, style As Long)
    With rng.Borders(edge)
        .LineStyle = style
        .Weight = xlThin
    End With
End Sub

'--- 印刷：A4たて1枚 ---
Private Sub 印刷の設定(ws As Worksheet, lastRow As Long)
    On Error Resume Next
    Application.PrintCommunication = False
    With ws.PageSetup
        .PrintArea = ws.Range(ws.Cells(1, 1), ws.Cells(lastRow, 10)).Address
        .Orientation = xlPortrait
        .PaperSize = xlPaperA4
        .Zoom = False
        .FitToPagesWide = 1
        .FitToPagesTall = 1
        .CenterHorizontally = True
        .TopMargin = Application.CentimetersToPoints(1.5)
        .BottomMargin = Application.CentimetersToPoints(1.5)
        .LeftMargin = Application.CentimetersToPoints(1.5)
        .RightMargin = Application.CentimetersToPoints(1.5)
    End With
    Application.PrintCommunication = True
    On Error GoTo 0
End Sub
